SERVICES
--------------------------------------------------------------------------------

1. gRPC API Service:
   {{- if contains "LoadBalancer" .Values.api.service.type }}
   export SERVICE_IP=$(kubectl get svc {{ include "ticketing.fullname" . }}-api -n {{ .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
   echo "API Endpoint: $SERVICE_IP:{{ .Values.api.service.port }}"
   {{- else if contains "ClusterIP" .Values.api.service.type }}
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "ticketing.fullname" . }}-api {{ .Values.api.service.port }}:{{ .Values.api.service.port }}
   echo "API Endpoint: localhost:{{ .Values.api.service.port }}"
   {{- end }}

2. Metrics Endpoint:
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "ticketing.fullname" . }}-api {{ .Values.api.service.metricsPort }}:{{ .Values.api.service.metricsPort }}
   curl http://localhost:{{ .Values.api.service.metricsPort }}/metrics

3. Health Checks:
   kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "ticketing.fullname" . }}-api {{ .Values.api.service.metricsPort }}:{{ .Values.api.service.metricsPort }}
   curl http://localhost:{{ .Values.api.service.metricsPort }}/health/ready
